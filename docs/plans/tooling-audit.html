! powershell -NoProfile -c "[Environment]::SetEnvironmentVariable('GITHUB_PERSONAL_ACCESS_TOKEN', (gh auth token), 'User')"

! cd ~/suryavanshi-claude && git config credential.helper "" && git config --add credential.helper manager && git config credential.useHttpPath true && git push
